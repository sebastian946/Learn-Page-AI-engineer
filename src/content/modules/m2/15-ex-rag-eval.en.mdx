```python
# 06-evals/golden.py
from __future__ import annotations

import json
from pathlib import Path
from typing import Literal

from pydantic import BaseModel, Field


class GoldenItem(BaseModel):
    id: str
    question: str
    reference: str
    expected_doc_id: str | None = None      # None for unanswerable items
    expected_section: str | None = None
    category: Literal["exact_term", "filtered", "paraphrase", "follow_up", "general"] = "general"
    filters: dict = Field(default_factory=dict)
    history: list[dict] = Field(default_factory=list)
    reference: str
    difficulty: Literal["lookup", "multi_hop", "numeric"]
    answerable: bool = True


def load_golden(path: Path = Path("data/golden.jsonl")) -> list[GoldenItem]:
    items = [GoldenItem.model_validate_json(line) for line in path.read_text("utf-8").splitlines() if line.strip()]
    ids = [i.id for i in items]
    assert len(ids) == len(set(ids)), "duplicate ids in golden set"
    return items


class RunRecord(BaseModel):
    item_id: str
    answer: str
    answer_found: bool
    retrieved_sources: list[tuple[str, str]]   # (doc_id, section), best-first
    retrieved_texts: list[str]
    latency_ms: float


def save_run(records: list[RunRecord], path: Path) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text("\n".join(r.model_dump_json() for r in records), "utf-8")
```
