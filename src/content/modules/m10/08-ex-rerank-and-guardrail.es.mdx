## Objetivo

Dos integraciones, cada una medida contra lo que ya tienes:

1. **Re-ranker.** Agrega un re-ranker con Noul de Jev a `rag-lab` como una carpeta nueva `08-jev-rerank/`, córrelo sobre el mismo `golden_rerank.jsonl` que usaste en `05-reranking/` del módulo 2, y compara hit@5 y MRR contra el cross-encoder, con latencia y costo por query.
2. **Gate de tool calls.** Agrega un nodo `jev_gate` a `langgraph-hitl-agent` que corra antes del nodo de aprobación y retenga llamadas irreversibles o fuera de tarea, y pruébalo con tests a nivel de grafo usando un modelo scripted y respuestas de Jev grabadas.

Ambas piezas van a sus repos originales. En `jev-lab`, la carpeta `03-integrations/` guarda solo el reporte de comparación y los links.

## Contexto

Para el re-ranker necesitas los artefactos del módulo 2: un retriever BM25 o híbrido que devuelva 30 candidatos por query, el archivo golden con el id del chunk relevante por query y los números del `bench.py` del cross-encoder. Si tu archivo golden tiene menos de 40 queries, extiéndelo a 40 primero; el MRR sobre 15 queries es ruido.

Para el gate necesitas el agente del módulo 4 con su fake `ScriptedModel` y el nodo de aprobación con `RISKY_TOOLS`. El gate no reemplaza ese nodo; decide si la llamada llega a él, lo salta o se rechaza.

# rag-lab/08-jev-rerank/reranker.py
from __future__ import annotations

import hashlib
import json
from concurrent.futures import ThreadPoolExecutor
from dataclasses import dataclass, field
from pathlib import Path

from typesafe_sdk import Noul, NoulCriteria, TypeSafeClient

RELEVANT = Noul(
    instructions="Does `passage` contain information that directly answers `query`?",
    criteria=NoulCriteria(
        true="The passage states the fact, policy, figure or procedure the query asks for.",
        false="The passage is about a related topic but does not answer the query.",
    ),
)


@dataclass
class Candidate:
    chunk_id: str
    title: str
    text: str
    bm25_score: float
    noul: float | None = None
    meta: dict = field(default_factory=dict)


class JevReranker:
    def __init__(self, client: TypeSafeClient, model: str = "jev-1.14.0", cache_dir: str = "cache", max_workers: int = 8):
        self.client, self.model, self.max_workers = client, model, max_workers
        self.cache = Path(cache_dir) / model
        self.cache.mkdir(parents=True, exist_ok=True)

    def _key(self, query: str, chunk_id: str) -> Path:
        return self.cache / (hashlib.sha256(f"{query}\n{chunk_id}".encode()).hexdigest() + ".json")

    def _score(self, query: str, c: Candidate) -> float:
        path = self._key(query, c.chunk_id)
        if path.exists():
            return json.loads(path.read_text("utf-8"))["noul"]
        r = self.client.system_one(
            state={"query": query, "passage": {"title": c.title, "text": c.text}},
            questions={"relevant": RELEVANT},
            model=self.model,
        )
        noul = r.answers["relevant"].noul
        path.write_text(json.dumps({"noul": noul, "model": r.model, "input_tokens": r.usage.input_tokens}), "utf-8")
        return noul

    def rerank(self, query: str, candidates: list[Candidate], top_k: int = 5) -> list[Candidate]:
        with ThreadPoolExecutor(max_workers=self.max_workers) as pool:
            scores = list(pool.map(lambda c: self._score(query, c), candidates))
        for c, s in zip(candidates, scores):
            c.noul = s
        # TODO(you): sort by noul desc, tie-break by bm25_score, return top_k
        raise NotImplementedError

from __future__ import annotations

from typing import Literal, Protocol

from typesafe_sdk import Choice, Noul

Decision = Literal["execute", "hold", "reject"]

GATE_QUESTIONS = {
    "reversible": Choice(
        instructions="If `tool_call` is executed, can its effect be undone without cost?",
        criteria={
            "reversible": "Read-only, or trivially undone by the system",
            "costly": "Undoing needs a person, a refund or a customer contact",
            "irreversible": "Cannot be undone (money sent, message delivered, data deleted)",
        },
    ),
    "on_task": Noul(instructions="Is `tool_call` a reasonable next step toward `goal` given `history`?"),
    "user_authorised": Noul(instructions="In `history`, did the user explicitly ask for the action in `tool_call`?"),
}


class JevLike(Protocol):
    def system_one(self, *, state, questions, model: str): ...


def build_state(agent_state: dict) -> dict:
    call = agent_state["pending_tool_call"]
    return {
        "goal": agent_state["goal"],
        "history": [m.content for m in agent_state["messages"][-6:]],
        "tool_call": {"name": call["name"], "args": call["args"]},
    }


def decide(answers: dict, cfg: dict, tool_name: str, risky_tools: set[str]) -> Decision:
    rev = answers["reversible"]
    on_task = answers["on_task"].noul
    authorised = answers["user_authorised"].noul
    if on_task < cfg["reject_if_on_task_below"]:
        return "reject"
    if tool_name in risky_tools:
        return "hold"                                    # static floor from Module 4
    # TODO(you): execute only if reversible with confidence >= cfg["execute_min_confidence"]
    #            and on_task >= cfg["execute_min_on_task"]; otherwise hold
    raise NotImplementedError


def jev_gate(agent_state: dict, client: JevLike, cfg: dict, risky_tools: set[str]) -> Decision:
    try:
        r = client.system_one(state=build_state(agent_state), questions=GATE_QUESTIONS, model=cfg["model"])
    except Exception:                                    # 529 after retries, timeout, network
        return "hold"                                    # fail closed
    return decide(r.answers, cfg, agent_state["pending_tool_call"]["name"], risky_tools)
